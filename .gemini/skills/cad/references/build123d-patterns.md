# build123d API Patterns Reference

Quick-reference for build123d code patterns used by the CAD skill.
The agent should follow these patterns when generating build123d scripts.

## Core imports

```python
from build123d import *
from build123d import export_step, export_stl, export_brep
```

Always import from `build123d`. Never use raw `OCP` calls or `cadquery`.

## Basic solid part

```python
from build123d import *

with BuildPart() as part:
    Box(60, 40, 10)

export_step(part.part, "box.step")
print(f"Volume: {part.part.volume:.1f} mm³")
print(f"Bounding box: {part.part.bounding_box()}")
```

## Sketch → Extrude workflow

```python
from build123d import *

with BuildPart() as part:
    with BuildSketch() as sk:
        Rectangle(60, 40)
        with Locations((20, 10), (-20, -10)):
            Circle(3, mode=Mode.SUBTRACT)
    extrude(amount=10)

export_step(part.part, "plate_with_holes.step")
```

## L-bracket pattern

```python
from build123d import *

with BuildPart() as part:
    # Horizontal leg
    Box(50, 30, 3)
    # Vertical leg
    with Locations((25 - 1.5, 0, 1.5)):
        Box(3, 30, 50, align=(Align.CENTER, Align.CENTER, Align.MIN))
    # Inner fillet for strength
    fillet(
        part.edges().filter_by(Axis.Y).sort_by(Axis.X)[-1:],
        radius=3,
    )

export_step(part.part, "l_bracket.step")
```

## Holes

```python
# Through-hole
with Locations((10, 0)):
    Hole(radius=2.25)  # M4 clearance = Ø4.5mm → radius 2.25

# Counterbore
with Locations((10, 0)):
    CounterBoreHole(
        radius=2.25,        # M4 clearance
        counter_bore_radius=4.5,  # Socket head cap
        counter_bore_depth=4,
    )

# Countersink
with Locations((10, 0)):
    CounterSinkHole(
        radius=1.7,          # M3 clearance
        counter_sink_radius=3.5,
        counter_sink_angle=82,
    )
```

## Shell (hollow out)

```python
with BuildPart() as part:
    Box(60, 40, 30)
    # Hollow out, keeping 2mm walls, opening the top face
    top_face = part.faces().sort_by(Axis.Z)[-1]
    shell(top_face, thickness=2)
```

## Fillet and chamfer

```python
# Fillet specific edges
fillet(part.edges().filter_by(Axis.Z), radius=2)

# Chamfer top edges
chamfer(part.edges().sort_by(Axis.Z)[-4:], length=1)

# Fillet all edges (use with care)
fillet(part.edges(), radius=0.5)
```

## Cylindrical parts

```python
with BuildPart() as part:
    Cylinder(radius=20, height=30)
    # Central bore
    Hole(radius=5)
    # Flange
    with Locations((0, 0, -15)):
        Cylinder(radius=30, height=5, align=(Align.CENTER, Align.CENTER, Align.MIN))
```

## Multiple positions

```python
# Regular grid
with GridLocations(x_spacing=20, y_spacing=20, x_count=3, y_count=2):
    Hole(radius=1.7)

# Polar array
with PolarLocations(radius=25, count=6):
    Hole(radius=2.25)

# Explicit positions
with Locations((10, 10), (-10, 10), (10, -10), (-10, -10)):
    Hole(radius=1.7)
```

## Mirror and symmetry

```python
with BuildPart() as part:
    Box(30, 20, 10)
    with Locations((10, 0)):
        Hole(radius=2)
    mirror(about=Plane.YZ)  # Mirror features across YZ plane
```

## Assemblies

```python
from build123d import *

# Part 1: Base plate
with BuildPart() as base:
    Box(100, 60, 5)
    with Locations((30, 0), (-30, 0)):
        Hole(radius=2.25)

# Part 2: Bracket (use code from above)
with BuildPart() as bracket:
    Box(3, 30, 40)

# Create assembly
assy = Compound(
    label="bracket_assembly",
    children=[
        Compound(label="base", wrapped=base.part.wrapped),
        Compound(
            label="bracket_left",
            wrapped=bracket.part.moved(Location((-30, 0, 5))).wrapped,
        ),
        Compound(
            label="bracket_right",
            wrapped=bracket.part.moved(Location((30, 0, 5))).wrapped,
        ),
    ],
)

export_step(assy, "bracket_assembly.step")
```

## Common pitfalls

### ❌ Don't do this

```python
# Wrong: using cadquery
import cadquery as cq
result = cq.Workplane("XY").box(10, 10, 10)

# Wrong: using raw OCP
from OCP.BRepPrimAPI import BRepPrimAPI_MakeBox

# Wrong: forgetting context manager
Box(10, 10, 10)  # This does nothing outside BuildPart

# Wrong: zero-thickness geometry
shell(face, thickness=0)

# Wrong: fillet radius larger than shortest edge
fillet(part.edges(), radius=50)  # Will fail on short edges
```

### ✅ Do this instead

```python
# Correct: always use build123d
from build123d import *

with BuildPart() as part:
    Box(10, 10, 10)
    
export_step(part.part, "output.step")
```

## Output template

Every generated script should follow this structure:

```python
"""
{part_name} — generated by AICAD
Description: {brief description}
Dimensions: {key dimensions}
"""
from build123d import *

# --- Parameters ---
LENGTH = 60   # mm
WIDTH = 40    # mm
HEIGHT = 10   # mm
WALL = 2      # mm

# --- Build ---
with BuildPart() as part:
    # ... geometry code ...
    pass

# --- Export ---
result = part.part
export_step(result, "{part_name}.step")

# --- Report ---
bb = result.bounding_box()
print(f"Part: {part_name}")
print(f"Bounding box: {bb.size.X:.1f} × {bb.size.Y:.1f} × {bb.size.Z:.1f} mm")
print(f"Volume: {result.volume:.1f} mm³")
print(f"Exported: {part_name}.step")
```
